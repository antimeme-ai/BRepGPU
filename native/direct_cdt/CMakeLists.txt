cmake_minimum_required(VERSION 3.28)
project(brepgpu_direct_cdt LANGUAGES CXX)
include(../../tooling/native_cpu_rigor.cmake)
add_library(brepgpu_direct_cdt SHARED
  direct_cdt.cpp core.cpp predicates.cpp ../face_domain/face_domain.cpp)
target_include_directories(brepgpu_direct_cdt PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../face_domain)
brepgpu_native_cpu_rigor(brepgpu_direct_cdt)
include(CTest)
foreach(fixture boundary predicates cap)
  add_executable(brepgpu_direct_cdt_${fixture} ../../tests/native_direct_cdt_${fixture}.cpp)
  target_link_libraries(brepgpu_direct_cdt_${fixture} PRIVATE brepgpu_direct_cdt)
  brepgpu_native_cpu_rigor(brepgpu_direct_cdt_${fixture})
  add_test(NAME native_direct_cdt_${fixture} COMMAND brepgpu_direct_cdt_${fixture})
  set_tests_properties(native_direct_cdt_${fixture} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(brepgpu_face_domain_check ../../tests/native_face_domain_check.cpp)
target_link_libraries(brepgpu_face_domain_check PRIVATE brepgpu_direct_cdt)
brepgpu_native_cpu_rigor(brepgpu_face_domain_check)
add_test(NAME native_face_domain COMMAND brepgpu_face_domain_check)
add_executable(brepgpu_face_domain_arithmetic ../../tests/native_face_domain_arithmetic.cpp)
target_link_libraries(brepgpu_face_domain_arithmetic PRIVATE brepgpu_direct_cdt)
brepgpu_native_cpu_rigor(brepgpu_face_domain_arithmetic)
