cmake_minimum_required(VERSION 3.28)
project(BRepGPUOM10ReferenceGeometry LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
option(BREPGPU_REFERENCE_GEOMETRY_SANITIZERS "Enable host address/undefined sanitizers" OFF)
add_executable(brepgpu-om10-reference-geometry main.cpp)
target_compile_features(brepgpu-om10-reference-geometry PRIVATE cxx_std_17)
target_include_directories(brepgpu-om10-reference-geometry PRIVATE ${OpenCASCADE_INCLUDE_DIR} ../../include)
target_link_directories(brepgpu-om10-reference-geometry PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(brepgpu-om10-reference-geometry PRIVATE TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
target_compile_options(brepgpu-om10-reference-geometry PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
if(UNIX AND NOT APPLE)
  target_link_options(brepgpu-om10-reference-geometry PRIVATE -Wl,--disable-new-dtags)
endif()
set_target_properties(brepgpu-om10-reference-geometry PROPERTIES BUILD_RPATH "${OpenCASCADE_LIBRARY_DIR}")
if(BREPGPU_REFERENCE_GEOMETRY_SANITIZERS)
  target_compile_options(brepgpu-om10-reference-geometry PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(brepgpu-om10-reference-geometry PRIVATE -fsanitize=address,undefined)
endif()
add_executable(brepgpu-reference-geometry-fixture ../../tests/reference_geometry_fixture.cpp)
target_compile_features(brepgpu-reference-geometry-fixture PRIVATE cxx_std_17)
target_include_directories(brepgpu-reference-geometry-fixture PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(brepgpu-reference-geometry-fixture PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(brepgpu-reference-geometry-fixture PRIVATE TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
target_compile_options(brepgpu-reference-geometry-fixture PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
if(UNIX AND NOT APPLE)
  target_link_options(brepgpu-reference-geometry-fixture PRIVATE -Wl,--disable-new-dtags)
endif()
set_target_properties(brepgpu-reference-geometry-fixture PROPERTIES BUILD_RPATH "${OpenCASCADE_LIBRARY_DIR}")
