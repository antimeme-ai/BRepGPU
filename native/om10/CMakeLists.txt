cmake_minimum_required(VERSION 3.28)
project(BRepGPUOM10 LANGUAGES CXX)
option(BREPGPU_OM10_ENABLE_CUDA "Build optional OM10 CUDA consumer" OFF)
option(BREPGPU_OM10_SANITIZERS "Enable host address/undefined sanitizers" OFF)
find_package(Threads REQUIRED)
include(CTest)
add_executable(brepgpu-om10-native main.cpp cpu.cpp)
target_include_directories(brepgpu-om10-native PRIVATE ../../include)
target_compile_features(brepgpu-om10-native PRIVATE cxx_std_17)
target_compile_options(brepgpu-om10-native PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(brepgpu-om10-native PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(BREPGPU_OM10_SANITIZERS)
 target_compile_options(brepgpu-om10-native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(brepgpu-om10-native PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
 add_executable(brepgpu-source-display-api ../../tests/native_mesher_display_api.cpp cpu.cpp)
 target_include_directories(brepgpu-source-display-api PRIVATE ../../include)
 target_compile_features(brepgpu-source-display-api PRIVATE cxx_std_17)
 target_compile_options(brepgpu-source-display-api PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
 target_link_libraries(brepgpu-source-display-api PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
 if(BREPGPU_OM10_SANITIZERS)
  target_compile_options(brepgpu-source-display-api PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(brepgpu-source-display-api PRIVATE -fsanitize=address,undefined)
 endif()
 add_test(NAME source_display_api COMMAND brepgpu-source-display-api)
 # Compiles cpu.cpp into the test so it can drive the production bin builder.
 add_executable(brepgpu-om10-cpu-bins ../../tests/native_om10_cpu_bins.cpp)
 target_include_directories(brepgpu-om10-cpu-bins PRIVATE ../../include)
 target_compile_features(brepgpu-om10-cpu-bins PRIVATE cxx_std_17)
 target_compile_options(brepgpu-om10-cpu-bins PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
 target_link_libraries(brepgpu-om10-cpu-bins PRIVATE Threads::Threads)
 if(BREPGPU_OM10_SANITIZERS)
  target_compile_options(brepgpu-om10-cpu-bins PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(brepgpu-om10-cpu-bins PRIVATE -fsanitize=address,undefined)
 endif()
 add_test(NAME cpu_bins_canonical_threads COMMAND brepgpu-om10-cpu-bins 1 2 3 12 24 48 96)
 if(BREPGPU_OM10_ENABLE_CUDA)
  add_test(NAME source_display_api_p100_0 COMMAND brepgpu-source-display-api $<TARGET_FILE:brepgpu_om10_cuda> 0)
  add_test(NAME source_display_api_p100_1 COMMAND brepgpu-source-display-api $<TARGET_FILE:brepgpu_om10_cuda> 1)
 endif()
endif()
if(BREPGPU_OM10_ENABLE_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 add_library(brepgpu_om10_cuda SHARED cuda.cu)
 target_include_directories(brepgpu_om10_cuda PRIVATE ../../include)
 target_link_libraries(brepgpu_om10_cuda PRIVATE CUDA::cudart)
 set_target_properties(brepgpu_om10_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 60-real CUDA_RUNTIME_LIBRARY Shared BUILD_RPATH "$<TARGET_FILE_DIR:CUDA::cudart>")
 target_compile_options(brepgpu_om10_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
 if(BUILD_TESTING)
  # Compiles cuda.cu into the test so it can drive the production bin builder.
  add_executable(brepgpu-om10-cuda-bins ../../tests/native_om10_cuda_bins.cu)
  target_include_directories(brepgpu-om10-cuda-bins PRIVATE ../../include)
  target_link_libraries(brepgpu-om10-cuda-bins PRIVATE CUDA::cudart)
  set_target_properties(brepgpu-om10-cuda-bins PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 60-real CUDA_RUNTIME_LIBRARY Shared BUILD_RPATH "$<TARGET_FILE_DIR:CUDA::cudart>")
  target_compile_options(brepgpu-om10-cuda-bins PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=false;--prec-div=true;--prec-sqrt=true;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  add_test(NAME cuda_bins_canonical_p100_0_1 COMMAND brepgpu-om10-cuda-bins 0 1)
 endif()
endif()
