cmake_minimum_required(VERSION 3.28)
project(BRepGPUOM10NvidiaReference LANGUAGES CXX)
set(BREPGPU_NVIDIA_PACKAGE "" CACHE PATH "Intact extracted official USD/BRep 1.0.0 package")
if(NOT EXISTS "${BREPGPU_NVIDIA_PACKAGE}/lib/libOCCT_BREP_IMPORT.a")
  message(FATAL_ERROR "BREPGPU_NVIDIA_PACKAGE must name the official native release extraction")
endif()
add_executable(brepgpu-om10-nvidia main.cpp)
add_library(brepgpu-om10-nvidia-raw STATIC raw_mesh.cpp)
add_executable(brepgpu-om10-nvidia-polygon-fixture EXCLUDE_FROM_ALL ../../tests/om10_nvidia_polygon_fixture.cpp)
foreach(target brepgpu-om10-nvidia brepgpu-om10-nvidia-raw brepgpu-om10-nvidia-polygon-fixture)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(component SMLib SM_API SM_API_USD)
    target_include_directories(${target} SYSTEM PRIVATE "${BREPGPU_NVIDIA_PACKAGE}/include/${component}")
  endforeach()
endforeach()
foreach(target brepgpu-om10-nvidia brepgpu-om10-nvidia-polygon-fixture)
  target_link_directories(${target} PRIVATE "${BREPGPU_NVIDIA_PACKAGE}/lib" "${BREPGPU_NVIDIA_PACKAGE}/extraLibs")
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${BREPGPU_NVIDIA_PACKAGE}/lib;${BREPGPU_NVIDIA_PACKAGE}/extraLibs")
endforeach()
target_link_libraries(brepgpu-om10-nvidia-polygon-fixture PRIVATE brepgpu-om10-nvidia-raw SM_API SMLib)
target_link_libraries(brepgpu-om10-nvidia PRIVATE brepgpu-om10-nvidia-raw OCCT_BREP_IMPORT BREP_STAGING BREP_USD_DATA SM_API_USD BREP_SM_USD USD_BREP_UTILS SM_API SMLib)
foreach(component ar arch gf js kind pcp plug python sdf sdr tf trace usd usdGeom usdLux usdShade usdUtils usdVol vt work)
  target_link_libraries(brepgpu-om10-nvidia PRIVATE "usd_${component}")
endforeach()
target_link_libraries(brepgpu-om10-nvidia PRIVATE tbb python3.12 dl)
