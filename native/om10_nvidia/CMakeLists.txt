cmake_minimum_required(VERSION 3.28)
project(BRepGPUOM10NvidiaReference LANGUAGES CXX)
set(BREPGPU_NVIDIA_PACKAGE "" CACHE PATH "Intact extracted official USD/BRep 1.0.0 package")
if(NOT EXISTS "${BREPGPU_NVIDIA_PACKAGE}/lib/libOCCT_BREP_IMPORT.a")
  message(FATAL_ERROR "BREPGPU_NVIDIA_PACKAGE must name the official native release extraction")
endif()
add_executable(brepgpu-om10-nvidia main.cpp)
target_compile_features(brepgpu-om10-nvidia PRIVATE cxx_std_17)
target_compile_options(brepgpu-om10-nvidia PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
foreach(component SMLib SM_API SM_API_USD)
  target_include_directories(brepgpu-om10-nvidia SYSTEM PRIVATE "${BREPGPU_NVIDIA_PACKAGE}/include/${component}")
endforeach()
target_link_directories(brepgpu-om10-nvidia PRIVATE "${BREPGPU_NVIDIA_PACKAGE}/lib" "${BREPGPU_NVIDIA_PACKAGE}/extraLibs")
target_link_libraries(brepgpu-om10-nvidia PRIVATE OCCT_BREP_IMPORT BREP_STAGING BREP_USD_DATA SM_API_USD BREP_SM_USD USD_BREP_UTILS SM_API SMLib)
foreach(component ar arch gf js kind pcp plug python sdf sdr tf trace usd usdGeom usdLux usdShade usdUtils usdVol vt work)
  target_link_libraries(brepgpu-om10-nvidia PRIVATE "usd_${component}")
endforeach()
target_link_libraries(brepgpu-om10-nvidia PRIVATE tbb python3.12 dl)
set_target_properties(brepgpu-om10-nvidia PROPERTIES BUILD_RPATH "${BREPGPU_NVIDIA_PACKAGE}/lib;${BREPGPU_NVIDIA_PACKAGE}/extraLibs")
